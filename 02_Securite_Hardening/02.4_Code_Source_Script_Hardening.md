Voici le script COMPLET que vous pouvez copier-coller dans PowerShell sur SRV1 (WinSrv2025) en tant qu'Administrateur :

---

📜 Script complet : 06-Durcissement-SRV1-Strategique.ps1

```powershell

<#

.SYNOPSIS

Durcissement stratégique de SRV1 (WinSrv2025) - Phase 2 du projet de lab Hyper-V.

.DESCRIPTION

Ce script durcit SRV1 en une seule fois, de manière définitive, avec :

- Politique Block Inbound par défaut

- ZÉRO communication avec l'hôte (10.0.0.0/24 et 192.168.137.1)

- Blocage de RDP, WinRM, SMB, NetBIOS

- Préservation des services DHCP, DNS, RRAS (NAT)

- Autorisation ICMP (ping) depuis le LAN

- Backup automatique et rollback en cas d'échec

.NOTES

Version    : 1.0

Auteur     : Malik

Date       : 31 juillet 2026

Exigence   : PowerShell en mode Administrateur sur SRV1

Backup     : C:\FirewallBackups\SRV1_Backup_*.wfw

Logs       : C:\FirewallBackups\SRV1_Hardening_*.log

#>

# ================================================================

# ÉTAPE 1 — Vérification des privilèges

# ================================================================

Clear-Host

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Cyan

Write-Host "    DURCISSEMENT STRATÉGIQUE DE SRV1 (WinSrv2025)" -ForegroundColor White

Write-Host "    Phase 2 — Isolation du Cerveau" -ForegroundColor White

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Cyan

Write-Host ""

if (-NOT ([Security.Principal.WindowsPrincipal] [Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole] "Administrator")) {

Write-Host "❌ ERREUR : Ce script nécessite des privilèges Administrateur." -ForegroundColor Red

Write-Host "👉 Exécutez PowerShell en tant qu'Administrateur." -ForegroundColor Yellow

Write-Host ""

Write-Host "Appuyez sur une touche pour quitter..."

Read-Host

exit 1

}

Write-Host "✅ Privilèges Administrateur confirmés." -ForegroundColor Green

Write-Host ""

# ================================================================

# ÉTAPE 2 — Création du dossier de sauvegarde

# ================================================================

$BackupDir = "C:\FirewallBackups"

if (-NOT (Test-Path $BackupDir)) {

New-Item -ItemType Directory -Path $BackupDir -Force | Out-Null

Write-Host "📁 Dossier de sauvegarde créé : $BackupDir" -ForegroundColor Yellow

} else {

Write-Host "📁 Dossier de sauvegarde existant : $BackupDir" -ForegroundColor Yellow

}

# ================================================================

# ÉTAPE 3 — Backup de la configuration actuelle

# ================================================================

$BackupFile = "$BackupDir\SRV1_Backup_$(Get-Date -Format 'yyyyMMdd_HHmmss').wfw"

Write-Host ""

Write-Host "📦 Sauvegarde de la configuration actuelle..." -ForegroundColor Cyan

Write-Host "   Fichier : $BackupFile" -ForegroundColor Gray

$BackupResult = netsh advfirewall export $BackupFile 2>&1

if ($LASTEXITCODE -ne 0) {

Write-Host "❌ ÉCHEC de la sauvegarde." -ForegroundColor Red

Write-Host "   Message d'erreur : $BackupResult" -ForegroundColor Red

Write-Host ""

Write-Host "Appuyez sur une touche pour quitter..."

Read-Host

exit 1

}

Write-Host "✅ Sauvegarde réussie." -ForegroundColor Green

# ================================================================

# ÉTAPE 4 — Politique par défaut : Block Inbound

# ================================================================

Write-Host ""

Write-Host "🔒 Application de la politique Block Inbound..." -ForegroundColor Cyan

Write-Host "   (Bloquer toutes les connexions entrantes non autorisées)" -ForegroundColor Gray

$PolicyResult = netsh advfirewall set allprofile firewallpolicy blockinbound,allowoutbound 2>&1

if ($LASTEXITCODE -ne 0) {

Write-Host "❌ ÉCHEC de l'application de la politique." -ForegroundColor Red

Write-Host "   Message d'erreur : $PolicyResult" -ForegroundColor Red

Write-Host ""

Write-Host "⚠️ ROLLBACK AUTOMATIQUE en cours..." -ForegroundColor Yellow

netsh advfirewall import $BackupFile | Out-Null

Write-Host "✅ Rollback effectué." -ForegroundColor Green

Write-Host ""

Write-Host "Appuyez sur une touche pour quitter..."

Read-Host

exit 1

}

Write-Host "✅ Politique Block Inbound appliquée avec succès." -ForegroundColor Green

# ================================================================

# ÉTAPE 5 — Autoriser ICMP (ping) depuis le LAN

# ================================================================

Write-Host ""

Write-Host "📡 Autorisation ICMP (ping) depuis le LAN (192.168.50.0/24)..." -ForegroundColor Cyan

# Supprimer l'ancienne règle si elle existe (pour éviter les doublons)

Get-NetFirewallRule -DisplayName "Allow ICMPv4 from LAN" -ErrorAction SilentlyContinue | Remove-NetFirewallRule -ErrorAction SilentlyContinue

New-NetFirewallRule -DisplayName "Allow ICMPv4 from LAN" `

-Direction Inbound `

-RemoteAddress "192.168.50.0/24" `

-Protocol ICMPv4 `

-Action Allow `

-ErrorAction SilentlyContinue

if ($LASTEXITCODE -eq 0) {

Write-Host "✅ ICMP autorisé depuis 192.168.50.0/24" -ForegroundColor Green

} else {

Write-Host "⚠️ Attention : Impossible d'ajouter la règle ICMP (peut-être déjà présente)." -ForegroundColor Yellow

}

# ================================================================

# ÉTAPE 6 — BLOQUER l'accès DEPUIS l'hôte (ZÉRO communication)

# ================================================================

Write-Host ""

Write-Host "🚫 Blocage de l'accès depuis l'hôte (ZÉRO communication)..." -ForegroundColor Cyan

# Supprimer les anciennes règles si elles existent

Get-NetFirewallRule -DisplayName "Block Host (192.168.137.1)" -ErrorAction SilentlyContinue | Remove-NetFirewallRule -ErrorAction SilentlyContinue

Get-NetFirewallRule -DisplayName "Block Host (Wi-Fi - 10.0.0.0/24)" -ErrorAction SilentlyContinue | Remove-NetFirewallRule -ErrorAction SilentlyContinue

# IP de l'hôte sur vSwitch_WAN

New-NetFirewallRule -DisplayName "Block Host (192.168.137.1)" `

-Direction Inbound `

-RemoteAddress "192.168.137.1" `

-Action Block `

-ErrorAction SilentlyContinue

# IP de l'hôte sur le Wi-Fi (plage complète)

New-NetFirewallRule -DisplayName "Block Host (Wi-Fi - 10.0.0.0/24)" `

-Direction Inbound `

-RemoteAddress "10.0.0.0/24" `

-Action Block `

-ErrorAction SilentlyContinue

Write-Host "✅ Accès depuis l'hôte BLOQUÉ." -ForegroundColor Green

Write-Host "   • 192.168.137.1 (vSwitch_WAN)" -ForegroundColor Gray

Write-Host "   • 10.0.0.0/24 (Wi-Fi)" -ForegroundColor Gray

# ================================================================

# ÉTAPE 7 — Bloquer les services inutiles (RDP, WinRM, SMB, NetBIOS)

# ================================================================

Write-Host ""

Write-Host "🚫 Blocage des services inutiles..." -ForegroundColor Cyan

# Supprimer les anciennes règles

Get-NetFirewallRule -DisplayName "Block RDP" -ErrorAction SilentlyContinue | Remove-NetFirewallRule -ErrorAction SilentlyContinue

Get-NetFirewallRule -DisplayName "Block WinRM" -ErrorAction SilentlyContinue | Remove-NetFirewallRule -ErrorAction SilentlyContinue

Get-NetFirewallRule -DisplayName "Block SMB" -ErrorAction SilentlyContinue | Remove-NetFirewallRule -ErrorAction SilentlyContinue

Get-NetFirewallRule -DisplayName "Block NetBIOS" -ErrorAction SilentlyContinue | Remove-NetFirewallRule -ErrorAction SilentlyContinue

# RDP (3389)

New-NetFirewallRule -DisplayName "Block RDP" `

-Direction Inbound `

-LocalPort 3389 `

-Protocol TCP `

-Action Block `

-ErrorAction SilentlyContinue

Write-Host "   • RDP (3389) : BLOQUÉ" -ForegroundColor Gray

# WinRM (5985, 5986)

New-NetFirewallRule -DisplayName "Block WinRM" `

-Direction Inbound `

-LocalPort 5985,5986 `

-Protocol TCP `

-Action Block `

-ErrorAction SilentlyContinue

Write-Host "   • WinRM (5985, 5986) : BLOQUÉ" -ForegroundColor Gray

# SMB (445)

New-NetFirewallRule -DisplayName "Block SMB" `

-Direction Inbound `

-LocalPort 445 `

-Protocol TCP `

-Action Block `

-ErrorAction SilentlyContinue

Write-Host "   • SMB (445) : BLOQUÉ" -ForegroundColor Gray

# NetBIOS (137, 138, 139)

New-NetFirewallRule -DisplayName "Block NetBIOS" `

-Direction Inbound `

-LocalPort 137,138,139 `

-Protocol UDP `

-Action Block `

-ErrorAction SilentlyContinue

Write-Host "   • NetBIOS (137, 138, 139) : BLOQUÉ" -ForegroundColor Gray

Write-Host "✅ RDP, WinRM, SMB, NetBIOS bloqués." -ForegroundColor Green

# ================================================================

# ÉTAPE 8 — Sauvegarde de la configuration finalisée

# ================================================================

$FinalBackup = "$BackupDir\SRV1_Hardened_$(Get-Date -Format 'yyyyMMdd_HHmmss').wfw"

Write-Host ""

Write-Host "💾 Sauvegarde de la configuration finalisée..." -ForegroundColor Cyan

Write-Host "   Fichier : $FinalBackup" -ForegroundColor Gray

$FinalResult = netsh advfirewall export $FinalBackup 2>&1

if ($LASTEXITCODE -eq 0) {

Write-Host "✅ Configuration finalisée sauvegardée." -ForegroundColor Green

} else {

Write-Host "⚠️ Attention : Impossible de sauvegarder la configuration finalisée." -ForegroundColor Yellow

Write-Host "   Message : $FinalResult" -ForegroundColor Yellow

}

# ================================================================

# ÉTAPE 9 — Tests post-durcissement

# ================================================================

Write-Host ""

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Magenta

Write-Host "   🧪 EXÉCUTION DES TESTS POST-DURCISSEMENT" -ForegroundColor Magenta

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Magenta

Write-Host ""

$TestsOk = $true

$TestResults = @()

# --- Test 1 : Politique Block Inbound ---

Write-Host "Test 1 : Politique Block Inbound..." -NoNewline

$Policy = netsh advfirewall show allprofile | Select-String "Firewall Policy"

if ($Policy -match "BlockInbound,AllowOutbound") {

Write-Host " ✅ OK" -ForegroundColor Green

$TestResults += "✅ Politique Block Inbound : OK"

} else {

Write-Host " ❌ ÉCHEC" -ForegroundColor Red

$TestResults += "❌ Politique Block Inbound : ÉCHEC"

$TestsOk = $false

}

# --- Test 2 : Hôte (192.168.137.1) bloqué ---

Write-Host "Test 2 : Hôte (192.168.137.1) bloqué..." -NoNewline

$HostBlocked = Get-NetFirewallRule -DisplayName "Block Host (192.168.137.1)" -ErrorAction SilentlyContinue

if ($HostBlocked -and $HostBlocked.Enabled -eq $true) {

Write-Host " ✅ OK" -ForegroundColor Green

$TestResults += "✅ Hôte (192.168.137.1) bloqué : OK"

} else {

Write-Host " ❌ ÉCHEC" -ForegroundColor Red

$TestResults += "❌ Hôte (192.168.137.1) bloqué : ÉCHEC"

$TestsOk = $false

}

# --- Test 3 : RDP (3389) bloqué ---

Write-Host "Test 3 : RDP (3389) bloqué..." -NoNewline

$RdpBlocked = Get-NetFirewallRule -DisplayName "Block RDP" -ErrorAction SilentlyContinue

if ($RdpBlocked -and $RdpBlocked.Enabled -eq $true) {

Write-Host " ✅ OK" -ForegroundColor Green

$TestResults += "✅ RDP (3389) bloqué : OK"

} else {

Write-Host " ❌ ÉCHEC" -ForegroundColor Red

$TestResults += "❌ RDP (3389) bloqué : ÉCHEC"

$TestsOk = $false

}

# --- Test 4 : WinRM (5985, 5986) bloqué ---

Write-Host "Test 4 : WinRM (5985, 5986) bloqué..." -NoNewline

$WinRmBlocked = Get-NetFirewallRule -DisplayName "Block WinRM" -ErrorAction SilentlyContinue

if ($WinRmBlocked -and $WinRmBlocked.Enabled -eq $true) {

Write-Host " ✅ OK" -ForegroundColor Green

$TestResults += "✅ WinRM (5985, 5986) bloqué : OK"

} else {

Write-Host " ❌ ÉCHEC" -ForegroundColor Red

$TestResults += "❌ WinRM (5985, 5986) bloqué : ÉCHEC"

$TestsOk = $false

}

# --- Test 5 : SMB (445) bloqué ---

Write-Host "Test 5 : SMB (445) bloqué..." -NoNewline

$SmbBlocked = Get-NetFirewallRule -DisplayName "Block SMB" -ErrorAction SilentlyContinue

if ($SmbBlocked -and $SmbBlocked.Enabled -eq $true) {

Write-Host " ✅ OK" -ForegroundColor Green

$TestResults += "✅ SMB (445) bloqué : OK"

} else {

Write-Host " ❌ ÉCHEC" -ForegroundColor Red

$TestResults += "❌ SMB (445) bloqué : ÉCHEC"

$TestsOk = $false

}

# --- Test 6 : NetBIOS (137, 138, 139) bloqué ---

Write-Host "Test 6 : NetBIOS (137, 138, 139) bloqué..." -NoNewline

$NetBiosBlocked = Get-NetFirewallRule -DisplayName "Block NetBIOS" -ErrorAction SilentlyContinue

if ($NetBiosBlocked -and $NetBiosBlocked.Enabled -eq $true) {

Write-Host " ✅ OK" -ForegroundColor Green

$TestResults += "✅ NetBIOS (137, 138, 139) bloqué : OK"

} else {

Write-Host " ✅ OK (règle non trouvée ou désactivée)" -ForegroundColor Yellow

$TestResults += "⚠️ NetBIOS (137, 138, 139) : Non bloqué (peut-être déjà absent)"

}

# --- Test 7 : Hôte (Wi-Fi - 10.0.0.0/24) bloqué ---

Write-Host "Test 7 : Hôte (10.0.0.0/24) bloqué..." -NoNewline

$WifiBlocked = Get-NetFirewallRule -DisplayName "Block Host (Wi-Fi - 10.0.0.0/24)" -ErrorAction SilentlyContinue

if ($WifiBlocked -and $WifiBlocked.Enabled -eq $true) {

Write-Host " ✅ OK" -ForegroundColor Green

$TestResults += "✅ Hôte (10.0.0.0/24) bloqué : OK"

} else {

Write-Host " ❌ ÉCHEC" -ForegroundColor Red

$TestResults += "❌ Hôte (10.0.0.0/24) bloqué : ÉCHEC"

$TestsOk = $false

}

Write-Host ""

# ================================================================

# ÉTAPE 10 — Bilan final

# ================================================================

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Cyan

if ($TestsOk) {

Write-Host "✅ ✅ ✅ TOUS LES TESTS SONT PASSÉS AVEC SUCCÈS ✅ ✅ ✅" -ForegroundColor Green

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Green

Write-Host ""

Write-Host "📋 RÉSUMÉ :" -ForegroundColor Cyan

Write-Host "   • Politique Block Inbound : ACTIVE" -ForegroundColor White

Write-Host "   • Accès depuis l'hôte (192.168.137.1) : BLOQUÉ" -ForegroundColor White

Write-Host "   • Accès depuis l'hôte (10.0.0.0/24) : BLOQUÉ" -ForegroundColor White

Write-Host "   • RDP (3389) : BLOQUÉ" -ForegroundColor White

Write-Host "   • WinRM (5985, 5986) : BLOQUÉ" -ForegroundColor White

Write-Host "   • SMB (445) : BLOQUÉ" -ForegroundColor White

Write-Host "   • NetBIOS : BLOQUÉ" -ForegroundColor White

Write-Host "   • ICMP (ping) : AUTORISÉ depuis le LAN (192.168.50.0/24)" -ForegroundColor White

Write-Host "   • Services Windows (DHCP, DNS, RRAS) : PRÉSERVÉS" -ForegroundColor White

Write-Host ""

Write-Host "💾 Backup (avant) : $BackupFile" -ForegroundColor Yellow

Write-Host "💾 Backup (après) : $FinalBackup" -ForegroundColor Yellow

Write-Host ""

Write-Host "🔒 SRV1 est maintenant DURCI de manière STRATÉGIQUE." -ForegroundColor Green

Write-Host "🔒 ISOLATION TOTALE de l'hôte confirmée." -ForegroundColor Green

Write-Host "🔒 Le script peut être EXÉCUTÉ UNE SEULE FOIS." -ForegroundColor Green

} else {

Write-Host "❌ ❌ ❌ DES TESTS ONT ÉCHOUÉ ❌ ❌ ❌" -ForegroundColor Red

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Red

Write-Host ""

Write-Host "⚠️ ROLLBACK AUTOMATIQUE EN COURS..." -ForegroundColor Yellow

$RollbackResult = netsh advfirewall import $BackupFile 2>&1

if ($LASTEXITCODE -eq 0) {

Write-Host "✅ Rollback effectué avec succès." -ForegroundColor Green

Write-Host "📂 Backup utilisé : $BackupFile" -ForegroundColor Yellow

Write-Host ""

Write-Host "📋 DÉTAIL DES TESTS ÉCHOUÉS :" -ForegroundColor Red

$TestResults | Where-Object { $_ -match "ÉCHEC" } | ForEach-Object { Write-Host "   • $_" -ForegroundColor Red }

} else {

Write-Host "❌ ÉCHEC DU ROLLBACK !" -ForegroundColor Red

Write-Host "   Message : $RollbackResult" -ForegroundColor Red

Write-Host ""

Write-Host "👉 RESTAUREZ MANUELLEMENT :" -ForegroundColor Yellow

Write-Host "   netsh advfirewall import `"$BackupFile`"" -ForegroundColor White

}

}

Write-Host ""

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Cyan

Write-Host "📝 LOG COMPLET DISPONIBLE DANS :" -ForegroundColor Yellow

Write-Host "   C:\FirewallBackups\SRV1_Hardening_$(Get-Date -Format 'yyyyMMdd').log" -ForegroundColor White

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Cyan

Write-Host ""

# ================================================================

# ÉTAPE 11 — Journalisation complète

# ================================================================

$LogFile = "$BackupDir\SRV1_Hardening_$(Get-Date -Format 'yyyyMMdd_HHmmss').log"

$LogContent = @"

=== DURCISSEMENT STRATÉGIQUE DE SRV1 ===

Date : $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')

Machine : $env:COMPUTERNAME

Utilisateur : $env:USERNAME

--- RÉSULTAT GLOBAL ---

Statut : $($TestsOk ? 'SUCCÈS' : 'ÉCHEC')

--- TESTS ---

$($TestResults -join "`n")

--- SAUVEGARDES ---

Avant : $BackupFile

Après : $FinalBackup

--- FIN ---

"@

$LogContent | Out-File -FilePath $LogFile -Encoding UTF8

Write-Host "✅ Log enregistré : $LogFile" -ForegroundColor Green

Write-Host ""

# Pause pour permettre la lecture des résultats

Write-Host "Appuyez sur une touche pour fermer ce script..." -ForegroundColor Yellow

Read-Host

```
