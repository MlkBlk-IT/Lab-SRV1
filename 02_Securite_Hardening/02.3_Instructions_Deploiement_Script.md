```powershell

# ============================================================

# CRÉATION DES FICHIERS DU PROJET SRV1 HARDENING

# ============================================================

# Créer le dossier

New-Item -ItemType Directory -Path "C:\SRV1-Hardening" -Force | Out-Null

Set-Location "C:\SRV1-Hardening"

# 1. Créer le script PowerShell

@'

# ================================================================

# 06-Durcissement-SRV1-Strategique.ps1

# ================================================================

# Objectif : Durcir SRV1 (WinSrv2025) en 1 seule fois, de manière

#            définitive, avec isolation TOTALE de l'hôte.

#

# Principe : Blocage au niveau des PORTS, pas des SERVICES.

#            Les services Windows gèrent leurs propres règles.

#

# Exigence : ZÉRO communication hôte ↔ SRV1/VMs

# ================================================================

# ÉTAPE 1 — Vérification des privilèges

if (-NOT ([Security.Principal.WindowsPrincipal] [Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole] "Administrator")) {

Write-Host "❌ Ce script nécessite des privilèges Administrateur." -ForegroundColor Red

Write-Host "👉 Exécutez PowerShell en tant qu'Administrateur." -ForegroundColor Yellow

exit 1

}

Write-Host "✅ Privilèges Administrateur confirmés." -ForegroundColor Green

Write-Host ""

# ÉTAPE 2 — Backup

$BackupDir = "C:\FirewallBackups"

if (-NOT (Test-Path $BackupDir)) {

New-Item -ItemType Directory -Path $BackupDir -Force | Out-Null

}

$BackupFile = "$BackupDir\SRV1_Backup_$(Get-Date -Format 'yyyyMMdd_HHmmss').wfw"

Write-Host "📦 Sauvegarde..." -ForegroundColor Cyan

netsh advfirewall export $BackupFile

if ($LASTEXITCODE -ne 0) {

Write-Host "❌ Échec de la sauvegarde. Abandon." -ForegroundColor Red

exit 1

}

Write-Host "✅ Sauvegarde : $BackupFile" -ForegroundColor Green

Write-Host ""

# ÉTAPE 3 — Politique Block Inbound

Write-Host "🔒 Application de la politique Block Inbound..." -ForegroundColor Cyan

netsh advfirewall set allprofile firewallpolicy blockinbound,allowoutbound

if ($LASTEXITCODE -ne 0) {

Write-Host "❌ Échec. Rollback..." -ForegroundColor Red

netsh advfirewall import $BackupFile

exit 1

}

Write-Host "✅ Politique Block Inbound appliquée." -ForegroundColor Green

Write-Host ""

# ÉTAPE 4 — Autoriser ICMP depuis le LAN

Write-Host "📡 Autorisation ICMP..." -ForegroundColor Cyan

New-NetFirewallRule -DisplayName "Allow ICMPv4 from LAN" -Direction Inbound -RemoteAddress "192.168.50.0/24" -Protocol ICMPv4 -Action Allow -ErrorAction SilentlyContinue

Write-Host "✅ ICMP autorisé." -ForegroundColor Green

Write-Host ""

# ÉTAPE 5 — Bloquer l'hôte (ZÉRO communication)

Write-Host "🚫 Blocage de l'hôte..." -ForegroundColor Cyan

New-NetFirewallRule -DisplayName "Block Host (192.168.137.1)" -Direction Inbound -RemoteAddress "192.168.137.1" -Action Block -ErrorAction SilentlyContinue

New-NetFirewallRule -DisplayName "Block Host (Wi-Fi - 10.0.0.0/24)" -Direction Inbound -RemoteAddress "10.0.0.0/24" -Action Block -ErrorAction SilentlyContinue

Write-Host "✅ Hôte bloqué." -ForegroundColor Green

Write-Host ""

# ÉTAPE 6 — Bloquer RDP, WinRM, SMB, NetBIOS

Write-Host "🚫 Blocage des services inutiles..." -ForegroundColor Cyan

New-NetFirewallRule -DisplayName "Block RDP" -Direction Inbound -LocalPort 3389 -Protocol TCP -Action Block -ErrorAction SilentlyContinue

New-NetFirewallRule -DisplayName "Block WinRM" -Direction Inbound -LocalPort 5985,5986 -Protocol TCP -Action Block -ErrorAction SilentlyContinue

New-NetFirewallRule -DisplayName "Block SMB" -Direction Inbound -LocalPort 445 -Protocol TCP -Action Block -ErrorAction SilentlyContinue

New-NetFirewallRule -DisplayName "Block NetBIOS" -Direction Inbound -LocalPort 137,138,139 -Protocol UDP -Action Block -ErrorAction SilentlyContinue

Write-Host "✅ Services bloqués." -ForegroundColor Green

Write-Host ""

# ÉTAPE 7 — Sauvegarde finale

$FinalBackup = "$BackupDir\SRV1_Hardened_$(Get-Date -Format 'yyyyMMdd_HHmmss').wfw"

Write-Host "💾 Sauvegarde finale..." -ForegroundColor Cyan

netsh advfirewall export $FinalBackup

Write-Host "✅ Final : $FinalBackup" -ForegroundColor Green

Write-Host ""

# ÉTAPE 8 — Tests

Write-Host "🧪 TESTS POST-DURCISSEMENT" -ForegroundColor Magenta

$TestsOk = $true

Write-Host "Test 1 : Politique Block Inbound..." -NoNewline

$Policy = netsh advfirewall show allprofile | Select-String "Firewall Policy"

if ($Policy -match "BlockInbound,AllowOutbound") { Write-Host " ✅ OK" -ForegroundColor Green } else { Write-Host " ❌ ÉCHEC" -ForegroundColor Red; $TestsOk = $false }

Write-Host "Test 2 : Hôte (192.168.137.1) bloqué..." -NoNewline

if (Get-NetFirewallRule -DisplayName "Block Host (192.168.137.1)" -ErrorAction SilentlyContinue) { Write-Host " ✅ OK" -ForegroundColor Green } else { Write-Host " ❌ ÉCHEC" -ForegroundColor Red; $TestsOk = $false }

Write-Host "Test 3 : RDP (3389) bloqué..." -NoNewline

if (Get-NetFirewallRule -DisplayName "Block RDP" -ErrorAction SilentlyContinue) { Write-Host " ✅ OK" -ForegroundColor Green } else { Write-Host " ❌ ÉCHEC" -ForegroundColor Red; $TestsOk = $false }

Write-Host "Test 4 : WinRM (5985,5986) bloqué..." -NoNewline

if (Get-NetFirewallRule -DisplayName "Block WinRM" -ErrorAction SilentlyContinue) { Write-Host " ✅ OK" -ForegroundColor Green } else { Write-Host " ❌ ÉCHEC" -ForegroundColor Red; $TestsOk = $false }

Write-Host "Test 5 : SMB (445) bloqué..." -NoNewline

if (Get-NetFirewallRule -DisplayName "Block SMB" -ErrorAction SilentlyContinue) { Write-Host " ✅ OK" -ForegroundColor Green } else { Write-Host " ❌ ÉCHEC" -ForegroundColor Red; $TestsOk = $false }

Write-Host ""

if ($TestsOk) {

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Green

Write-Host "✅ ✅ ✅ TOUS LES TESTS SONT PASSÉS AVEC SUCCÈS ✅ ✅ ✅" -ForegroundColor Green

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Green

Write-Host ""

Write-Host "🔒 SRV1 est maintenant DURCI de manière STRATÉGIQUE." -ForegroundColor Green

Write-Host "🔒 ISOLATION TOTALE de l'hôte confirmée." -ForegroundColor Green

Write-Host ""

Write-Host "💾 Backup : $BackupFile" -ForegroundColor Yellow

Write-Host "💾 Final : $FinalBackup" -ForegroundColor Yellow

} else {

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Red

Write-Host "❌ ❌ ❌ DES TESTS ONT ÉCHOUÉ ❌ ❌ ❌" -ForegroundColor Red

Write-Host "═══════════════════════════════════════════════════════════" -ForegroundColor Red

Write-Host ""

Write-Host "👉 ROLLBACK AUTOMATIQUE..." -ForegroundColor Yellow

netsh advfirewall import $BackupFile

Write-Host "✅ Rollback effectué." -ForegroundColor Green

}

'@ | Out-File -FilePath "06-Durcissement-SRV1-Strategique.ps1" -Encoding UTF8
```
